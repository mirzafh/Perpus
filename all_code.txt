    _namaController.dispose();

  void _login() async {
  final inputNama = _namaController.text.trim();
  final inputNisn = _passwordController.text.trim(); // Diinput di password (berisi NISN)

  if (inputNama.isEmpty || inputNisn.isEmpty) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Nama dan NISN (password) wajib diisi!')),
    );
    return;
  }

  // 1. Cek Login Admin
  if (inputNama == '1' && inputNisn == '1') {
    Navigator.pushReplacement(
      context,
      MaterialPageRoute(
        builder: (context) => Homeadmins(
          onToggleTheme: widget.onToggleTheme,
        ),
      ),
    );
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Berhasil Login Sebagai Admin')),
    );
    return;
  }

  // 2. Cek Login Siswa Berdasarkan Document ID (NISN)
  try {
    // Mengambil dokumen langsung menggunakan NISN sebagai ID-nya
    final docSnapshot = await FirebaseFirestore.instance
        .collection('murid')
        .doc(inputNisn)
        .get();

    if (docSnapshot.exists) {
      final dataMurid = docSnapshot.data() as Map<String, dynamic>;
      final namaDatabase = (dataMurid['nama'] ?? '').toString().trim();

      // Cocokkan nama yang diinput dengan nama di database (case-insensitive)
      if (namaDatabase.toLowerCase() == inputNama.toLowerCase()) {
        if (!mounted) return;
        Navigator.pushReplacement(
          context,
          MaterialPageRoute(
            builder: (context) => HomeStudents(
              onToggleTheme: widget.onToggleTheme,
              nisn: inputNisn, // Teruskan NISN ke halaman siswa
            ),
          ),
        );
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Selamat datang, $namaDatabase!')),
        );
      } else {
        if (!mounted) return;
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Nama murid tidak sesuai dengan NISN tersebut!')),
        );
      }
    } else {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('NISN tidak terdaftar di database!')),
      );
    }
  } catch (e) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('Terjadi kesalahan: $e')),
    );
  }

class DataMuridAdmin extends StatefulWidget {
  const DataMuridAdmin({super.key});

  @override
  State<DataMuridAdmin> createState() => _DataMuridAdminState();
}

class _DataMuridAdminState extends State<DataMuridAdmin> {
  final TextEditingController searchController = TextEditingController();
  String searchQuery = '';

  final CollectionReference _muridRef =
      FirebaseFirestore.instance.collection('murid');

  @override
  void dispose() {
    searchController.dispose();
    super.dispose();
  }

  void _showMessage(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

  Future<void> addStudent() async {
    final result = await showDialog<Map<String, String>>(
      context: context,
      builder: (context) => const TambahMurid(),
    );

    if (result == null || !mounted) return;

    final nisn = result['nisn']!;
    try {
      final doc = await _muridRef.doc(nisn).get();
      if (doc.exists) {
        _showMessage('NISN sudah terdaftar!');
        return;
      }

      await _muridRef.doc(nisn).set({
        'nama': result['nama'],
        'nisn': nisn,
        'kelas': result['kelas'],
        'gender': result['gender'],
        'password': nisn,
        'streak': 0,
        'wishlist': <String>[],
        'createdAt': FieldValue.serverTimestamp(),
      });

      _showMessage('Murid berhasil ditambahkan.');
    } catch (e) {
      _showMessage('Gagal menambahkan murid: $e');
    }
  }

  Future<void> editStudent(Map<String, String> student) async {
    final oldNisn = student['nisn']!;
    final result = await showDialog<Map<String, String>>(
      context: context,
      builder: (context) => EditMurid(student: student),
    );

    if (result == null || !mounted) return;

    final newNisn = result['nisn']!;
    try {
      if (newNisn != oldNisn) {
        final cekDoc = await _muridRef.doc(newNisn).get();
        if (cekDoc.exists) {
          _showMessage('NISN tersebut sudah digunakan murid lain.');
          return;
        }
        await _muridRef.doc(oldNisn).delete();
      }

      await _muridRef.doc(newNisn).set({
        'nama': result['nama'],
        'nisn': newNisn,
        'kelas': result['kelas'],
        'gender': result['gender'],
        'password': newNisn,
      }, SetOptions(merge: true));

      _showMessage('Data murid berhasil diperbarui.');
    } catch (e) {
      _showMessage('Gagal memperbarui data: $e');
    }
  }

  Future<void> deleteStudent(Map<String, String> student) async {
    final nama = student['nama'] ?? 'murid ini';
    final nisn = student['nisn'];
    if (nisn == null) return;

    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => DialogHapusMurid(
        title: 'Hapus Data Murid?',
        message: 'Apakah kamu yakin ingin menghapus data $nama? '
            'Data yang dihapus tidak dapat dikembalikan.',
        confirmText: 'Hapus',
      ),
    );

    if (confirmed != true || !mounted) return;

    try {
      await _muridRef.doc(nisn).delete();
      _showMessage('Data $nama berhasil dihapus.');
    } catch (e) {
      _showMessage('Gagal menghapus data: $e');
    }
  }

  Future<void> deleteAllStudents(int currentCount) async {
    if (currentCount == 0) {
      _showMessage('Tidak ada data murid untuk dihapus.');
      return;
    }

    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => DialogHapusMurid(
        title: 'Hapus Semua Data Murid?',
        message: 'Seluruh $currentCount data murid akan dihapus. '
            'Tindakan ini tidak dapat dibatalkan.',
        confirmText: 'Hapus Semua',
        destructiveAll: true,
      ),
    );

      final batch = FirebaseFirestore.instance.batch();
      for (final doc in snapshot.docs) {
        batch.delete(doc.reference);
      }
      await batch.commit();
      _showMessage('Seluruh data murid berhasil dihapus.');
    } catch (e) {
      _showMessage('Gagal menghapus semua data: $e');
    }
  }

  Future<void> importExcel(Set<String> existingNisn) async {
    try {
      final pickedFile = await fp.FilePicker.pickFile(
        type: fp.FileType.custom,
        allowedExtensions: ['xlsx'],
      );

      if (pickedFile == null) return;
      final bytes = await pickedFile.readAsBytes();
      final workbook = xl.Excel.decodeBytes(bytes);

      if (workbook.tables.isEmpty) {
        throw Exception('File Excel kosong.');
      }

      final sheet = workbook.tables.values.first;
      final rows = sheet.rows;

      if (rows.length < 2) {
        throw Exception('Tidak ada data murid.');
      }

      String getCell(int row, int col) {
        if (col >= rows[row].length) return '';
        return rows[row][col]?.value?.toString().trim() ?? '';
      }

      final headers = rows.first.map((cell) {
        return cell?.value?.toString().trim().toLowerCase() ?? '';
      }).toList();

      int findColumn(List<String> aliases) {
        return headers.indexWhere((header) => aliases.contains(header));
      }

      final namaCol = findColumn(['nama', 'nama siswa']);
      final nisnCol = findColumn(['nisn']);
      final kelasCol = findColumn(['kelas']);
      final genderCol = findColumn(['jenis kelamin', 'gender', 'jenis_kelamin']);

      if ([namaCol, nisnCol, kelasCol, genderCol].contains(-1)) {
        throw Exception(
          'Header Excel harus berisi Nama, NISN, Kelas, Jenis Kelamin.',
        );
      }

      final batch = FirebaseFirestore.instance.batch();
      int importedCount = 0;
      int skipped = 0;

      for (int i = 1; i < rows.length; i++) {
        final nama = getCell(i, namaCol);
        final nisn = getCell(i, nisnCol);
        final kelas = getCell(i, kelasCol).toUpperCase();
        final genderRaw = getCell(i, genderCol).toLowerCase();

        if (nama.isEmpty && nisn.isEmpty) continue;

        if (nama.isEmpty || !RegExp(r'^\d{10}$').hasMatch(nisn)) {
          skipped++;
          continue;
        }

        if (!['X', 'XA', 'XB', 'XI', 'XII'].contains(kelas)) {
          skipped++;
          continue;
        }

        String gender;
        if (['l', 'laki-laki', 'laki laki'].contains(genderRaw)) {
          gender = 'Laki-laki';
        } else if (['p', 'perempuan'].contains(genderRaw)) {
          gender = 'Perempuan';
        } else {
          skipped++;
          continue;
        }

        if (existingNisn.contains(nisn)) {
          skipped++;
          continue;
        }

        existingNisn.add(nisn);
        final docRef = _muridRef.doc(nisn);
        batch.set(docRef, {
          'nama': nama,
          'nisn': nisn,
          'kelas': kelas,
          'gender': gender,
          'password': nisn,
          'streak': 0,
          'wishlist': <String>[],
          'createdAt': FieldValue.serverTimestamp(),
        });
        importedCount++;
      }

      if (importedCount > 0) {
        await batch.commit();
      }

      _showMessage(
        '$importedCount murid berhasil diimpor. $skipped baris dilewati.',
      );
    } catch (e) {
      _showMessage('Gagal mengimpor Excel: $e');
    }
  }

  @override
  void initState() {
    super.initState();
    _timer = Timer.periodic(const Duration(seconds: 1), (timer) {

  String formatWaktu(DateTime waktu) {
    String duaDigit(int angka) => angka.toString().padLeft(2, '0');
    return '${duaDigit(waktu.day)}/'
        '${duaDigit(waktu.month)}/'
        '${waktu.year} '
        '${duaDigit(waktu.hour)}:'
        '${duaDigit(waktu.minute)}:'
        '${duaDigit(waktu.second)}';
  }

  Future<void> prosesScan(String isbn) async {
    isbn = isbn.trim();
    if (isbn.isEmpty) return;

    final sudahAda = daftarBuku.any((buku) => buku['isbn'] == isbn);
    if (sudahAda) {
      tampilkanPesan('Buku ini sudah masuk daftar peminjaman');
      _scanController.clear();
      _scanFocus.requestFocus();
      return;
    }

    try {
      final doc = await FirebaseFirestore.instance
          .collection('buku')
          .doc(isbn)
          .get();

      if (!mounted) return;

      if (!doc.exists) {
        tampilkanPesan('ISBN "$isbn" tidak ditemukan di database buku');
        _scanController.clear();
        _scanFocus.requestFocus();
        return;
      }

      final data = doc.data() as Map<String, dynamic>;
      final stok = (data['stok'] is num) ? (data['stok'] as num).toInt() : 0;

      if (stok <= 0) {
        tampilkanPesan('Stok buku "${data['judul']}" sedang habis!');
        _scanController.clear();
        _scanFocus.requestFocus();
        return;
      }

      setState(() {
        daftarBuku.add({
          'isbn': isbn,
          'judul': (data['judul'] ?? '').toString(),
          'penerbit': (data['penerbit'] ?? '').toString(),
          'kategori': (data['kategori'] ?? 'Lainnya').toString(),
          'kode': '',
        });
      });
    } catch (e) {
      tampilkanPesan('Gagal mencari buku: $e');
    }

    _scanController.clear();
    _scanFocus.requestFocus();
  }

  Future<void> tampilkanPopupSiswa() async {

    if (hasil != null) {

  void konfirmasiPeminjaman() {
    if (namaSiswa == null || nisSiswa == null) {
      tampilkanPesan('Silakan pilih siswa terlebih dahulu');
      return;
    }
    if (deadline == null) {
      tampilkanPesan('Silakan tentukan deadline peminjaman');
      return;
    }
    if (daftarBuku.isEmpty) {
      tampilkanPesan('Belum ada buku yang dipindai');
      return;
    }

    final kodeKosong = daftarBuku.any(
      (buku) => buku['kode'].toString().trim().isEmpty,
    );
    if (kodeKosong) {
      tampilkanPesan('Semua kode inventaris harus diisi');
      return;
    }

    final kodeDuplikat = daftarBuku
            .map((buku) => buku['kode'].toString().trim())
            .toSet()
            .length !=
        daftarBuku.length;
    if (kodeDuplikat) {
      tampilkanPesan('Ada kode inventaris yang sama');
      return;
    }

    showDialog(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          title: const Text('Peminjaman Siap Disimpan'),
          content: Text(
            'Siswa: $namaSiswa ($nisSiswa)\n'
            'Jumlah buku: ${daftarBuku.length}\n'
            'Deadline: ${formatWaktu(deadline!)}',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),

            ),

                Navigator.pop(dialogContext);

              },
              child: const Text('Konfirmasi'),
            ),
          ],
        );
      },
    );
  }

  Future<void> _simpanKeFirebase() async {
    try {
      final db = FirebaseFirestore.instance;
      final batch = db.batch();
      final sekarang = Timestamp.now();
      final batas = Timestamp.fromDate(deadline!);

      for (final buku in daftarBuku) {
        final pinjamRef = db.collection('peminjaman').doc();
        batch.set(pinjamRef, {
          'nisn': nisSiswa,
          'namaSiswa': namaSiswa,
          'kelasSiswa': kelasSiswa ?? '-',
          'genderSiswa': genderSiswa ?? '-',
          'isbn': buku['isbn'],
          'judulBuku': buku['judul'],
          'penerbit': buku['penerbit'],
          'kategoriBuku': buku['kategori'],
          'kodeInventaris': buku['kode'].toString().trim(),
          'status': 'dipinjam',
          'waktuPinjam': sekarang,
          'deadline': batas,
          'waktuKembali': null,
        });

        // Kurangi stok buku & tambah dipinjamCount
        final bukuDoc = db.collection('buku').doc(buku['isbn']);
        batch.update(bukuDoc, {
          'stok': FieldValue.increment(-1),
          'dipinjamCount': FieldValue.increment(1),
        });

        // Buat notifikasi untuk siswa
        final notifRef = db.collection('notifikasi').doc();
        batch.set(notifRef, {
          'nisn': nisSiswa,
          'title': 'Peminjaman Berhasil',
          'subtitle':
              'Kamu berhasil meminjam buku "${buku['judul']}" (Kode: ${buku['kode']}).',
          'type': 'info',
          'waktu': sekarang,
          'dibaca': false,
        });
      }

      // Tambah streak meminjam pada dokumen murid
      final muridDoc = db.collection('murid').doc(nisSiswa);
      batch.set(
        muridDoc,
        {'streak': FieldValue.increment(1)},
        SetOptions(merge: true),
      );

      await batch.commit();

      if (!mounted) return;
      setState(() {
        daftarBuku.clear();
        namaSiswa = null;
        nisSiswa = null;
        deadline = null;
      });
      tampilkanPesan('Transaksi peminjaman berhasil disimpan ke Firebase!');
    } catch (e) {
      tampilkanPesan('Gagal menyimpan peminjaman: $e');
    }
  }

  String formatWaktu(DateTime? waktu) {
    if (waktu == null) return '-';
    String duaDigit(int angka) => angka.toString().padLeft(2, '0');
    return '${duaDigit(waktu.day)}/'
        '${duaDigit(waktu.month)}/'
        '${waktu.year} '
        '${duaDigit(waktu.hour)}:'
        '${duaDigit(waktu.minute)}';
  }

  void konfirmasiPengembalian(DocumentSnapshot doc) {
    final data = doc.data() as Map<String, dynamic>;
    if (data['status'] == 'kembali') return;

    showDialog(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          icon: const Icon(
            Icons.warning_amber_rounded,
            color: Colors.orange,
            size: 42,
          ),
          title: const Text('Konfirmasi Pengembalian'),
          content: Text(
            'Anda yakin ingin mengubah status peminjaman '
            'pada ${data['namaSiswa']} menjadi sudah dikembalikan?\n\n'
            'Buku: ${data['judulBuku']}\n'
            'Kode: ${data['kodeInventaris']}',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Batal'),
            ),
            FilledButton(
              onPressed: () async {
                Navigator.pop(dialogContext);
                try {
                  final db = FirebaseFirestore.instance;
                  final batch = db.batch();

                  batch.update(doc.reference, {
                    'status': 'kembali',
                    'waktuKembali': Timestamp.now(),
                  });

                  final isbn = (data['isbn'] ?? '').toString();
                  if (isbn.isNotEmpty) {
                    batch.update(db.collection('buku').doc(isbn), {
                      'stok': FieldValue.increment(1),
                    });
                  }

                  await batch.commit();

                  if (!mounted) return;
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(
                      content: Text('Buku berhasil dikonfirmasi kembali'),
                    ),
                  );
                } catch (e) {
                  if (!mounted) return;
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(content: Text('Gagal mengonfirmasi: $e')),
                  );
                }

              child: const Text('Ya, Konfirmasi'),
            ),
          ],
        );
      },

import 'package:flutter/material.dart';

class HasilKelolaKategori {
  final List<String> kategori;
  final Map<String, String> perubahanNama;

  HasilKelolaKategori({
    required this.kategori,
    required this.perubahanNama,
  });
}

class KelolaKategoriDialog extends StatefulWidget {
  final List<String> kategoriAwal;
  final Set<String> kategoriTerpakai;

  const KelolaKategoriDialog({
    super.key,
    required this.kategoriAwal,
    required this.kategoriTerpakai,
  });

  @override
  State<KelolaKategoriDialog> createState() =>
      _KelolaKategoriDialogState();
}

class _KelolaKategoriDialogState
    extends State<KelolaKategoriDialog> {
  late List<String> kategori;
  late Set<String> kategoriTerpakai;
  final TextEditingController controller = TextEditingController();
  final Map<String, String> perubahanNama = {};

  @override
  void initState() {
    super.initState();
    kategori = List.from(widget.kategoriAwal);
    kategoriTerpakai = Set.from(widget.kategoriTerpakai);
  }

  @override
  void dispose() {
    controller.dispose();
    super.dispose();
  }

  void tampilkanPesan(String pesan) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(pesan)),
    );
  }

  void tambahKategori() {
    final nama = controller.text.trim();

    if (nama.isEmpty) {
      tampilkanPesan('Nama kategori tidak boleh kosong');
      return;
    }

    final sudahAda = kategori.any(
      (item) => item.toLowerCase() == nama.toLowerCase(),
    );

    if (sudahAda) {
      tampilkanPesan('Kategori sudah tersedia');
      return;
    }

    setState(() {
      kategori.add(nama);
      controller.clear();
    });
  }

  Future<void> editKategori(String namaLama) async {
    final editController = TextEditingController(text: namaLama);

    final namaBaru = await showDialog<String>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('Edit Kategori'),
        content: TextField(
          controller: editController,
          autofocus: true,
          decoration: const InputDecoration(
            labelText: 'Nama kategori',
            border: OutlineInputBorder(),
          ),
          onSubmitted: (value) {
            Navigator.pop(dialogContext, value.trim());
          },
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext),
            child: const Text('Batal'),
          ),
          FilledButton(
            onPressed: () {
              Navigator.pop(
                dialogContext,
                editController.text.trim(),
              );
            },
            child: const Text('Simpan'),
          ),
        ],
      ),
    );

    editController.dispose();

    if (!mounted || namaBaru == null || namaBaru.isEmpty) {
      return;
    }

    final sudahAda = kategori.any(
      (item) =>
          item.toLowerCase() == namaBaru.toLowerCase() &&
          item != namaLama,
    );

    if (sudahAda) {
      tampilkanPesan('Nama kategori sudah digunakan');
      return;
    }

    setState(() {
      final index = kategori.indexOf(namaLama);
      if (index == -1) return;

      kategori[index] = namaBaru;

      if (kategoriTerpakai.remove(namaLama)) {
        kategoriTerpakai.add(namaBaru);
      }

      final namaAsal = perubahanNama.remove(namaLama) ?? namaLama;
      perubahanNama[namaBaru] = namaAsal;
    });
  }

  void hapusKategori(String nama) {
    if (kategoriTerpakai.contains(nama)) {
      tampilkanPesan(
        'Kategori masih digunakan oleh buku',
      );
      return;
    }

    setState(() {
      kategori.remove(nama);
      perubahanNama.remove(nama);
    });
  }

  void simpanKategori() {
    Navigator.pop(
      context,
      HasilKelolaKategori(
        kategori: List.from(kategori),
        perubahanNama: Map.from(perubahanNama),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final colors = Theme.of(context).colorScheme;

    return AlertDialog(
      title: Row(
        children: [
          const Icon(Icons.category_outlined),
          const SizedBox(width: 10),
          const Expanded(child: Text('Kelola Kategori')),
        ],
      ),
      content: SizedBox(
        width: 420,
        height: 400,
        child: Column(
          children: [
            Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: controller,
                    onSubmitted: (_) => tambahKategori(),
                    decoration: InputDecoration(
                      hintText: 'Nama kategori baru',
                      filled: true,
                      fillColor: colors.surfaceContainerHighest,
                      border: OutlineInputBorder(
                        borderRadius: BorderRadius.circular(12),
                        borderSide: BorderSide.none,
                      ),
                    ),
                  ),
                ),
                const SizedBox(width: 8),
                IconButton.filled(
                  onPressed: tambahKategori,
                  icon: const Icon(Icons.add),
                  tooltip: 'Tambah kategori',
                ),
              ],
            ),
            const SizedBox(height: 16),
            Align(
              alignment: Alignment.centerLeft,
              child: Text(
                '${kategori.length} kategori terdaftar',
                style: TextStyle(color: colors.secondary),
              ),
            ),
            const SizedBox(height: 8),
            Expanded(
              child: kategori.isEmpty
                  ? const Center(
                      child: Text('Belum ada kategori'),
                    )
                  : ListView.separated(
                      itemCount: kategori.length,
                      separatorBuilder: (_, _) =>
                          const Divider(height: 1),
                      itemBuilder: (context, index) {
                        final nama = kategori[index];
                        final sedangDigunakan =
                            kategoriTerpakai.contains(nama);

                        return ListTile(
                          contentPadding:
                              const EdgeInsets.symmetric(horizontal: 4),
                          title: Text(nama),
                          subtitle: sedangDigunakan
                              ? const Text('Sedang digunakan')
                              : null,
                          trailing: Row(
                            mainAxisSize: MainAxisSize.min,
                            children: [
                              IconButton(
                                onPressed: () => editKategori(nama),
                                icon: const Icon(Icons.edit_outlined),
                                tooltip: 'Edit kategori',
                              ),
                              IconButton(
                                onPressed: () => hapusKategori(nama),
                                icon: Icon(
                                  Icons.delete_outline,
                                  color: sedangDigunakan
                                      ? colors.onSurface.withValues(
                                          alpha: 0.3,
                                        )
                                      : colors.error,
                                ),
                                tooltip: sedangDigunakan
                                    ? 'Kategori masih digunakan'
                                    : 'Hapus kategori',
                              ),
                            ],
                          ),
                        );
                      },
                    ),
            ),
          ],
        ),
      ),
      actions: [
        FilledButton(
          onPressed: simpanKategori,
          child: const Text('Selesai'),
        ),

  }

import 'package:flutter/material.dart';
import 'package:cloud_firestore/cloud_firestore.dart';

class PopupDataSiswa {
  static Future<Map<String, String>?> tampilkan(BuildContext context) {
    final cariController = TextEditingController();

    return showDialog<Map<String, String>>(
      context: context,
      builder: (dialogContext) {
        return StatefulBuilder(
          builder: (context, setPopupState) {
            return AlertDialog(
              title: const Text('Pilih Data Siswa'),
              content: SizedBox(
                width: 400,
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    TextField(
                      controller: cariController,
                      onChanged: (_) => setPopupState(() {}),
                      decoration: const InputDecoration(
                        hintText: 'Cari nama atau NISN siswa',
                        prefixIcon: Icon(Icons.search),
                        border: OutlineInputBorder(),
                      ),
                    ),
                    const SizedBox(height: 15),
                    SizedBox(
                      height: 220,
                      child: StreamBuilder<QuerySnapshot>(
                        stream: FirebaseFirestore.instance
                            .collection('murid')
                            .orderBy('nama')
                            .snapshots(),
                        builder: (context, snapshot) {
                          if (snapshot.connectionState ==
                              ConnectionState.waiting) {
                            return const Center(
                              child: CircularProgressIndicator(),
                            );
                          }

                          final docs = snapshot.data?.docs ?? [];
                          final daftarSiswa = docs.map((doc) {
                            final data = doc.data() as Map<String, dynamic>;
                            return {
                              'nama': (data['nama'] ?? '').toString(),
                              'nis': (data['nisn'] ?? doc.id).toString(),
                              'kelas': (data['kelas'] ?? '-').toString(),
                              'gender': (data['gender'] ?? '-').toString(),
                            };
                          }).toList();

                          final kata = cariController.text.toLowerCase();
                          final hasilPencarian = daftarSiswa.where((siswa) {
                            return siswa['nama']!.toLowerCase().contains(kata) ||
                                siswa['nis']!.contains(kata);
                          }).toList();

                          if (hasilPencarian.isEmpty) {
                            return const Center(
                              child: Text('Siswa tidak ditemukan'),
                            );
                          }

                          return ListView.builder(
                            itemCount: hasilPencarian.length,
                            itemBuilder: (context, index) {
                              final siswa = hasilPencarian[index];
                              return ListTile(
                                leading: const CircleAvatar(
                                  child: Icon(Icons.person),
                                ),
                                title: Text(siswa['nama']!),
                                subtitle: Text(
                                  'NISN: ${siswa['nis']} â€¢ Kelas ${siswa['kelas']}',
                                ),
                                onTap: () {
                                  Navigator.pop(dialogContext, siswa);
                                },
                              );
                            },
                          );
                        },
                      ),
                    ),
                  ],
                ),
              ),
              actions: [
                TextButton(
                  onPressed: () => Navigator.pop(dialogContext),
                  child: const Text('Batal'),
                ),
              ],
            );
          },
        );
      },
    ).whenComplete(() {
      cariController.dispose();
    });
  }
}

class Buku {
  final String isbn;
  final String judul;
  final String penerbit;
  final String kategori;
  final int stok;
  final Uint8List? cover; // Untuk preview sementara / data Base64 lama
  final String? coverUrl; // URL Firebase Storage
  final int dipinjamCount;

  const Buku({
    required this.isbn,
    required this.judul,
    required this.penerbit,
    required this.kategori,
    required this.stok,
    this.cover,
    this.coverUrl,
    this.dipinjamCount = 0,
  });

  Buku copyWith({
    String? isbn,
    String? judul,
    String? penerbit,
    String? kategori,
    int? stok,
    Uint8List? cover,
    String? coverUrl,
    int? dipinjamCount,
  }) {
    return Buku(
      isbn: isbn ?? this.isbn,
      judul: judul ?? this.judul,
      penerbit: penerbit ?? this.penerbit,
      kategori: kategori ?? this.kategori,
      stok: stok ?? this.stok,
      cover: cover ?? this.cover,
      coverUrl: coverUrl ?? this.coverUrl,
      dipinjamCount: dipinjamCount ?? this.dipinjamCount,
    );
  }

  Map<String, dynamic> toMap() {
    return {
      'isbn': isbn,
      'judul': judul,
      'penerbit': penerbit,
      'kategori': kategori,
      'stok': stok,
      'coverUrl': coverUrl,
      'dipinjamCount': dipinjamCount,
    };
  }

  factory Buku.fromMap(Map<String, dynamic> map, {String? docId}) {
    Uint8List? decodedCover;
    final coverStr = map['coverBase64'];
    if (coverStr is String && coverStr.isNotEmpty) {
      try {
        decodedCover = base64Decode(coverStr);
      } catch (_) {}
    }
    return Buku(
      isbn: (map['isbn'] ?? docId ?? '').toString(),
      judul: (map['judul'] ?? '').toString(),
      penerbit: (map['penerbit'] ?? '').toString(),
      kategori: (map['kategori'] ?? 'Lainnya').toString(),
      stok: (map['stok'] is num)
          ? (map['stok'] as num).toInt()
          : int.tryParse('${map['stok']}') ?? 0,
      cover: decodedCover,
      coverUrl: (map['coverUrl'] ?? '').toString().isEmpty
          ? null
          : map['coverUrl'].toString(),
      dipinjamCount: (map['dipinjamCount'] is num)
          ? (map['dipinjamCount'] as num).toInt()
          : 0,
    );
  }
}

  void submitData() async {
  final nama = namaController.text.trim();
  final nisn = nisnController.text.trim();

  if (nama.isEmpty ||
      nisn.isEmpty ||
      selectedKelas == null ||
      selectedGender == null) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Lengkapi semua data murid!')),
    );
    return;
  }

  if (!RegExp(r'^\d{10}$').hasMatch(nisn)) {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('NISN harus terdiri dari 10 angka!')),
    );
    return;
  }

  try {
    // 1. Cek apakah NISN sudah pernah terdaftar di database
    final cekNisn = await FirebaseFirestore.instance
        .collection('murid')
        .where('nisn', isEqualTo: nisn)
        .get();

    if (cekNisn.docs.isNotEmpty) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('NISN ini sudah terdaftar!')),
      );
      return;
    }

    // 2. Simpan murid baru ke koleksi 'murid'
    await FirebaseFirestore.instance.collection('murid').add({
      'nama': nama,
      'nisn': nisn,
      'kelas': selectedKelas,
      'gender': selectedGender,
      'createdAt': FieldValue.serverTimestamp(),

    if (!mounted) return;
    Navigator.pop(context, true);
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Data murid berhasil ditambahkan ke database!')),

import 'dart:typed_data';
import 'package:firebase_storage/firebase_storage.dart';

class UploadCover {
  static Future<String> upload({
    required String isbn,
    required Uint8List bytes,
  }) async {
    final ref = FirebaseStorage.instance
        .ref()
        .child('covers')
        .child('$isbn-${DateTime.now().millisecondsSinceEpoch}.jpg');

    await ref.putData(
      bytes,
      SettableMetadata(contentType: 'image/jpeg'),
    );
    return ref.getDownloadURL();
  }

  static Future<void> deleteByUrl(String? url) async {
    if (url == null || url.isEmpty) return;
    try {
      await FirebaseStorage.instance.refFromURL(url).delete();
    } on FirebaseException catch (e) {
      if (e.code != 'object-not-found') rethrow;
    }
  }
}

import 'package:perpus/pages/BookDetail.dart';
import 'package:perpus/widgets/admin/ModelBuku.dart';

class BookRowStudent extends StatelessWidget {
  final String title;
  final bool showTitle;
  final String? filterGenre;
  final bool sortByPopular;

  const BookRowStudent({
    super.key,
    required this.title,
    required this.showTitle,
    this.filterGenre,
    this.sortByPopular = false,
  });

  @override
  Widget build(BuildContext context) {
    final screenWidth = MediaQuery.of(context).size.width;

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        if (showTitle) ...[
          Padding(
            padding: EdgeInsets.symmetric(horizontal: screenWidth * 0.05),
            child: Text(
              title,
              style: TextStyle(
                fontSize: 16,
                fontWeight: FontWeight.w600,
                color: Colors.grey[700],
              ),
            ),
          ),
        ],
        const SizedBox(height: 12),
        SizedBox(
          height: 195,
          child: StreamBuilder<QuerySnapshot>(
            stream: FirebaseFirestore.instance.collection('buku').snapshots(),
            builder: (context, snapshot) {
              if (snapshot.connectionState == ConnectionState.waiting) {
                return const Center(child: CircularProgressIndicator());
              }

              var listBuku = (snapshot.data?.docs ?? []).map((doc) {
                return Buku.fromMap(
                  doc.data() as Map<String, dynamic>,
                  docId: doc.id,
                );
              }).toList();

              if (filterGenre != null && filterGenre != 'Semua') {
                listBuku = listBuku
                    .where((b) =>
                        b.kategori.toLowerCase() == filterGenre!.toLowerCase())
                    .toList();
              }

              if (sortByPopular) {
                listBuku.sort(
                  (a, b) => b.dipinjamCount.compareTo(a.dipinjamCount),
                );
              }

              if (listBuku.isEmpty) {
                return Padding(
                  padding: EdgeInsets.symmetric(horizontal: screenWidth * 0.05),
                  child: const Center(
                    child: Text('Belum ada buku pada kategori ini.'),
                  ),
                );
              }

              return ListView.builder(
                scrollDirection: Axis.horizontal,
                itemCount: listBuku.length,
                padding: EdgeInsets.symmetric(horizontal: screenWidth * 0.05),
                itemBuilder: (context, index) {
                  final buku = listBuku[index];
                  return GestureDetector(
                    onTap: () {
                      Navigator.push(
                        context,
                        MaterialPageRoute(
                          builder: (context) => DetailBook(buku: buku),
                        ),
                      );
                    },
                    child: Container(
                      width: 135,
                      margin: const EdgeInsets.only(right: 15),
                      decoration: BoxDecoration(
                        gradient: LinearGradient(
                          begin: Alignment.topCenter,
                          end: Alignment.bottomCenter,
                          colors: [
                            Colors.grey.shade300,
                            Colors.grey.shade400,
                          ],
                        ),
                        borderRadius: BorderRadius.circular(15),
                      ),
                      clipBehavior: Clip.antiAlias,
                      child: Stack(
                        fit: StackFit.expand,
                        children: [
                          if (buku.cover != null)
                            Image.memory(buku.cover!, fit: BoxFit.cover)
                          else if (buku.coverUrl != null &&
                              buku.coverUrl!.isNotEmpty)
                            Image.network(
                              buku.coverUrl!,
                              fit: BoxFit.cover,
                              errorBuilder: (_, __, ___) => const Center(
                                child: Icon(
                                  Icons.menu_book_rounded,
                                  color: Colors.black26,
                                  size: 46,
                                ),
                              ),
                            )
                          else
                            const Center(
                              child: Icon(
                                Icons.menu_book_rounded,
                                color: Colors.black26,
                                size: 46,
                              ),
                            ),
                          Align(
                            alignment: Alignment.bottomCenter,
                            child: Container(
                              width: double.infinity,
                              padding: const EdgeInsets.all(8),
                              color: Colors.black.withValues(alpha: 0.6),
                              child: Text(
                                buku.judul,
                                maxLines: 2,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(
                                  color: Colors.white,
                                  fontSize: 12,
                                  fontWeight: FontWeight.w600,
                                ),
                              ),
                            ),
                          ),
                        ],
                      ),
                    ),
                  );
                },
              );
            },
          ),
        ),
      ],
    );
  }
}

import 'package:flutter/material.dart';

class GenreBook extends StatelessWidget {
  final List<String> genres;
  final int selectedIndex;
  final Function(int) onTap;

  const GenreBook({
    super.key,
    required this.genres,
    required this.selectedIndex,
    required this.onTap,
  });

  @override
  Widget build(BuildContext context) {
    final screenWidth = MediaQuery.of(context).size.width;
    final colorScheme = Theme.of(context).colorScheme;
    return SizedBox(
      height: (screenWidth * 0.1).clamp(35.0, 50.0),
      child: ListView.builder(
        scrollDirection: Axis.horizontal,
        itemCount: genres.length,
        padding: EdgeInsets.symmetric(horizontal: screenWidth * 0.05),
        itemBuilder: (context, index) {
          bool isActive = index == selectedIndex;

          return GestureDetector(
            onTap: () => onTap(index),
            child: Container(
              margin: const EdgeInsets.only(right: 10),
              padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 8),
              decoration: BoxDecoration(
                color: isActive ? colorScheme.primary : colorScheme.surface,
                borderRadius: BorderRadius.circular(25),
                border: Border.all(
                  color: isActive ? colorScheme.primary : colorScheme.outline,
                  width: 1.5,
                ),
              ),
              child: Center(
                child: Text(
                  genres[index],
                  style: TextStyle(
                    color: isActive ? colorScheme.onPrimary : colorScheme.onSurfaceVariant,
                    fontWeight: isActive ? FontWeight.bold : FontWeight.normal,
                    fontSize: (screenWidth * 0.035).clamp(12.0, 16.0),
                  ),
                ),
              ),
            ),
          );
        },
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:cloud_firestore/cloud_firestore.dart';

class HeaderTopStudents extends StatelessWidget {
  final VoidCallback onToggleTheme;
  final String currentNisn;

  const HeaderTopStudents({
    super.key,
    required this.onToggleTheme,
    this.currentNisn = '10293', // NISN default siswa yang sedang aktif
  });

  // Menghitung total peminjaman siswa dari Firestore
  Future<int> _getTotalBukuDipinjam() async {
    try {
      final query = await FirebaseFirestore.instance
          .collection('peminjaman')
          .where('nisn', isEqualTo: currentNisn)
          .count()
          .get();
      return query.count ?? 0;
    } catch (e) {
      return 0;
    }
  }

  @override
  Widget build(BuildContext context) {
    final isDark = Theme.of(context).brightness == Brightness.dark;
    final screenWidth = MediaQuery.of(context).size.width;
    final colorBox = Theme.of(context).colorScheme.surface;
    final colorOutline = Theme.of(context).colorScheme.outline;
    final colorIcon = Theme.of(context).colorScheme.primary;

    return Container(
      height: MediaQuery.of(context).size.height * 0.1 > 90
          ? MediaQuery.of(context).size.height * 0.1
          : 90,
      decoration: BoxDecoration(
        gradient: LinearGradient(
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
          colors: [colorIcon, Colors.transparent],
        ),
      ),
      child: SafeArea(
        child: Padding(
          padding: EdgeInsets.symmetric(horizontal: screenWidth * 0.05),
          child: Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              // Ikon Logo Kiri
              Container(
                height: (screenWidth * 0.1).clamp(40, 60),
                width: (screenWidth * 0.1).clamp(40, 60),
                decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(100),
                  color: colorBox,
                  border: Border.all(color: colorOutline, width: 3.0),
                ),
                child: Center(
                  child: Icon(
                    Icons.book,
                    color: colorIcon,
                    size: (screenWidth * 0.06).clamp(24, 32),
                  ),
                ),
              ),

              // Bagian Total Buku Dipinjam (Menggantikan 9999 statis)
              Container(
                height: (screenWidth * 0.1).clamp(40, 60),
                padding: EdgeInsets.symmetric(
                  horizontal: screenWidth * 0.03,
                  vertical: 8,
                ),
                decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(30),
                  color: Theme.of(context).colorScheme.surface,
                  border: Border.all(color: colorOutline, width: 3.0),
                ),
                child: Center(
                  child: Row(
                    mainAxisSize: MainAxisSize.min,
                    children: [
                      Icon(
                        Icons.auto_stories_outlined,
                        color: const Color(0xFFD97706),
                        size: (screenWidth * 0.05).clamp(20, 28),
                      ),
                      SizedBox(width: screenWidth * 0.015),
                      FutureBuilder<int>(
                        future: _getTotalBukuDipinjam(),
                        builder: (context, snapshot) {
                          final count = snapshot.data ?? 0;
                          return Text(
                            '$count',
                            style: TextStyle(
                              color: const Color(0xFFD97706),
                              fontSize: (screenWidth * 0.04).clamp(14, 20),
                              fontWeight: FontWeight.bold,
                            ),
                          );
                        },
                      ),
                    ],
                  ),
                ),
              ),

              // Tombol Pengubah Tema
              InkWell(
                onTap: onToggleTheme,
                borderRadius: BorderRadius.circular(100),
                child: Container(
                  height: (screenWidth * 0.1).clamp(40, 60),
                  width: (screenWidth * 0.1).clamp(40, 60),
                  decoration: BoxDecoration(
                    borderRadius: BorderRadius.circular(100),
                    color: colorBox,
                    border: Border.all(
                      color: colorOutline,
                      width: 3.0,
                    ),
                  ),
                  child: Center(
                    child: Icon(
                      isDark ? Icons.dark_mode : Icons.sunny,
                      color: Theme.of(context).colorScheme.primary,
                      size: (screenWidth * 0.06).clamp(24, 32),
                    ),
